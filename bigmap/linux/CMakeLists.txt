cmake_minimum_required(VERSION 3.13)
project(tpf2_bigmap_linux LANGUAGES CXX ASM)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 17)
add_library(tpf2_bigmap MODULE bigmap_linux.cpp terrain_minmax.cpp terrain_minmax.S)
set_target_properties(tpf2_bigmap PROPERTIES PREFIX "" SUFFIX ".so")
target_compile_options(tpf2_bigmap PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
target_link_libraries(tpf2_bigmap PRIVATE Threads::Threads)
target_link_options(tpf2_bigmap PRIVATE -static-libstdc++ -static-libgcc -Wl,--exclude-libs,ALL)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_bigmap tests/bigmap_test.cpp tests/town_stub.S terrain_minmax.cpp terrain_minmax.S)
  target_compile_options(test_bigmap PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(test_bigmap PRIVATE Threads::Threads)
  add_test(NAME bigmap COMMAND test_bigmap)
endif()

add_executable(bigmap-density-restore density_restore.cpp)
target_link_options(bigmap-density-restore PRIVATE -static-libstdc++ -static-libgcc)
if(BUILD_TESTING)
  add_executable(test_density tests/density_test.cpp)
  target_compile_options(test_density PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_options(test_density PRIVATE -static-libstdc++ -static-libgcc)
  add_test(NAME density COMMAND test_density)
endif()

if(BUILD_TESTING)
  add_executable(test_pager tests/pager_test.cpp)
  target_compile_options(test_pager PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(test_pager PRIVATE Threads::Threads)
  add_test(NAME pager COMMAND test_pager)
  set_tests_properties(pager PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 40)
endif()

if(BUILD_TESTING)
  add_executable(test_terrain_codec ../tools/test_terrain_codec.cpp)
  add_test(NAME terrain_codec COMMAND test_terrain_codec)
endif()

# Export only the shared plugin ABI, including in a standalone build.
if(NOT CMAKE_MODULE_LINKER_FLAGS MATCHES "--version-script")
  target_link_options(tpf2_bigmap PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/../../native/linux/exports_plugin.map")
endif()
